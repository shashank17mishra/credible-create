💡 How to enable real-time GET reading in your Google Apps Script (Spreadsheet)

Your Google Apps Script URL is set in .env as APPS_SCRIPT_URL.

To allow your admin dashboard to fetch spreadsheet rows in real-time, ensure your Google Apps Script includes this doGet(e) handler:

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var result = [];
  
  for (var i = 1; i < data.length; i++) {
    var row = {};
    for (var j = 0; j < headers.length; j++) {
      row[headers[j]] = data[i][j];
    }
    result.push(row);
  }
  
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}
